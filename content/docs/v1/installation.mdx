## Install

Pin the minor so a stray `^` does not pull you onto v2.

```npm
npm install @letterstack/react@~1.4.0
```

## Provider

```tsx title="app/layout.tsx"
import '@letterstack/react/styles.css';
import { LetterStackProvider } from '@letterstack/react';

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html lang="en">
      <body>
        <LetterStackProvider>{children}</LetterStackProvider>
      </body>
    </html>
  );
}
```

The provider API did not change in v2, so this file needs no edits when you upgrade.
